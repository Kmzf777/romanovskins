PLANO DE IMPLEMENTAÇÃO TÉCNICA (PRD)ResumoPlataforma de rifas digitais focada em skins de CS2. O sistema utiliza uma arquitetura de "reserva otimista" onde cotas selecionadas ficam travadas por 20 minutos no banco de dados (PostgreSQL) para evitar vendas duplicadas (Race Conditions). A autenticação é simplificada (Mobile-First) via WhatsApp/Nome. O processamento financeiro é realizado via integração PIX/AbacatePay com confirmação via Webhooks.Stack TecnológicaRuntime: Node.js 20+ (Ambiente Serverless/Edge do Next.js)Framework: Next.js 15 (App Router, React Server Components)Estilização: Tailwind CSS + Shadcn/UI (para componentes rápidos)Banco de Dados: Supabase (PostgreSQL 15+)Gerenciamento de Estado (Client): Zustand (para o carrinho de compras/seleção de números)Pagamentos: AbacatePay APIAgendamento/Cron: Supabase pg_cron (para liberar reservas expiradas)Estrutura de PastasSnippet de códigosrc/

Schema SQL (Supabase)Este schema garante a integridade das reservas e evita vendas duplicadas.SQL-- Tabela de Usuários (Simplificada conforme pedido)
CREATE TABLE users (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    name TEXT NOT NULL,
    whatsapp TEXT NOT NULL UNIQUE, -- Chave de identificação
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabela de Rifas (Sorteios)
CREATE TABLE raffles (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    title TEXT NOT NULL,
    description TEXT,
    image_url TEXT NOT NULL,
    price_per_ticket DECIMAL(10, 2) NOT NULL,
    total_numbers INT NOT NULL, -- Ex: 100, 1000
    status TEXT DEFAULT 'active' CHECK (status IN ('active', 'closed', 'cancelled')),
    winner_number INT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabela de Cotas (A "alma" do sistema)
-- Status: 'available', 'reserved', 'sold'
CREATE TABLE tickets (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    raffle_id UUID REFERENCES raffles(id) ON DELETE CASCADE,
    ticket_number INT NOT NULL,
    status TEXT DEFAULT 'available' CHECK (status IN ('available', 'reserved', 'sold')),
    user_id UUID REFERENCES users(id), -- Null se available
    reserved_at TIMESTAMPTZ, -- Data da reserva
    expires_at TIMESTAMPTZ,  -- reserved_at + 20 min
    UNIQUE(raffle_id, ticket_number) -- Impede números duplicados na mesma rifa
);

-- Tabela de Transações (AbacatePay)
CREATE TABLE transactions (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID REFERENCES users(id),
    raffle_id UUID REFERENCES raffles(id),
    external_id TEXT, -- ID do AbacatePay
    amount DECIMAL(10, 2),
    status TEXT DEFAULT 'pending', -- pending, paid, failed
    ticket_numbers INT[], -- Array dos números comprados
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Índices para performance
CREATE INDEX idx_tickets_raffle_status ON tickets(raffle_id, status);
CREATE INDEX idx_tickets_expires_at ON tickets(expires_at) WHERE status = 'reserved';

Lógica Crítica / Algoritmos EspecíficosO Algoritmo de Reserva Atômica (Anti-Colisão)O maior risco é dois usuários comprarem o número "50" ao mesmo tempo. Resolvemos isso com SQL direto na Server Action. Não fazemos "Select depois Update", fazemos tudo em um comando.Arquivo: src/server/raffle-actions.tsTypeScriptimport { createClient } from '@/lib/supabase/server';

export async function reserveTickets(userId: string, raffleId: string, numbers: number[]) {
  const supabase = createClient();
  const expiresAt = new Date(Date.now() + 20 * 60 * 1000).toISOString(); // +20 min

  // Tenta atualizar SOMENTE se estiver disponível
  // Isso garante que o primeiro a chegar leva, o segundo falha
  const { data, error } = await supabase
    .from('tickets')
    .update({ 
      status: 'reserved', 
      user_id: userId,
      reserved_at: new Date().toISOString(),
      expires_at: expiresAt
    })
    .eq('raffle_id', raffleId)
    .in('ticket_number', numbers)
    .eq('status', 'available') // CRÍTICO: Só atualiza se ainda estiver livre
    .select();

  if (error) throw new Error('Erro ao reservar');

  // Verifica se conseguiu reservar TODOS os números solicitados
  if (data.length !== numbers.length) {
    // Se falhou algum, precisamos fazer rollback dos que conseguimos (opcional, mas recomendado)
    // Para MVP, avisamos o usuário que alguns números já foram pegos
    return { success: false, reserved: data, missing_count: numbers.length - data.length };
  }

  return { success: true, redirectUrl: `/checkout/${raffleId}` };
}
